import os

BACK=("127.0.0.1",21080); KC=("127.0.0.1",21081); HOSTH="localhost:21173"
def req(addr, method, path, headers=None, body=None, host=None):
    h={"Host": host or f"{addr[0]}:{addr[1]}"}; h.update(headers or {})
    c=http.client.HTTPConnection(*addr, timeout=30)
    c.request(method, path, body=body, headers=h)
    r=c.getresponse(); b=r.read().decode("utf8","replace")
    return r.status, r.getheaders(), b
def setcookies(hd):
    out={}
    for k,v in hd:
        if k.lower()=="set-cookie":
            n,_,rest=v.partition("="); out[n]=rest.split(";")[0]
    return out
